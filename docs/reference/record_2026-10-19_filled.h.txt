#pragma once

#include <cstddef>
#include <cstdint>
#include <cstring>
#include <string>

namespace lsmkv {

enum class RecordType : uint8_t {
  kPut = 1,
  kDelete = 2,
};

struct Record {
  RecordType type = RecordType::kPut;
  std::string key;
  std::string value;
};

inline void PutFixed32(std::string* dst, uint32_t v) {
  char buf[4];
  std::memcpy(buf, &v, 4);
  dst->append(buf, 4);
}

inline uint32_t GetFixed32(const char** p, const char* limit) {
  if (*p + 4 > limit) {
    return 0;
  }
  uint32_t v;
  std::memcpy(&v, *p, 4);
  *p += 4;
  return v;
}

// ── 你自己写的 Encode ──
inline std::string Encode(const Record& record) {
  std::string out;

  out.push_back(static_cast<char>(record.type));
  PutFixed32(&out, record.key.size());
  out.append(record.key);
  PutFixed32(&out, record.value.size());
  out.append(record.value);

  return out;
}

// ── 你自己写的 Decode（前两步是我给的范例，后三步你补的）──
inline bool Decode(const std::string& buf, Record* out, size_t* consumed) {
  const char* p = buf.data();
  const char* limit = buf.data() + buf.size();

  if (p >= limit) {
    return false;
  }
  out->type = static_cast<RecordType>(*p);
  ++p;

  if (p + 4 > limit) {
    return false;
  }
  const uint32_t klen = GetFixed32(&p, limit);

  if (p + klen > limit) {
    return false;
  }
  out->key.assign(p, klen);
  p += klen;

  if (p + 4 > limit) {
    return false;
  }
  const uint32_t vlen = GetFixed32(&p, limit);

  if (p + vlen > limit) {
    return false;
  }
  out->value.assign(p, vlen);
  p += vlen;

  if (consumed != nullptr) {
    *consumed = static_cast<size_t>(p - buf.data());
  }
  return true;
}

}  // namespace lsmkv
